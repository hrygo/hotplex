type audit987ThreadSink struct { manager *CodexAppServerManager }
func (s *audit987ThreadSink) Close() error { return nil }
func (s *audit987ThreadSink) Write(p []byte) (int, error) {
    var frame JSONRPCRequest
    if err := json.Unmarshal(p, &frame); err != nil { return 0, err }
    if frame.ID != 0 {
        if pending, ok := s.manager.pending.Load(frame.ID); ok {
            pending.(chan *JSONRPCResponse) <- &JSONRPCResponse{Result: json.RawMessage(`{"thread":{"id":"audit-old-thread"}}`)}
        }
    }
    return len(p), nil
}

func TestAudit987OwnershipResetFailureReleasesOnlyOwnReference(t *testing.T) {
    t.Parallel()
    m := auditRPCManager()
    m.stdin = &audit987ThreadSink{manager:m}
    m.refs = 1 // another active session; this Worker must not consume its reference
    w := &AppServerWorker{BaseWorker:base.NewBaseWorker(nil,nil), manager:m}
    require.NoError(t, w.Start(context.Background(),worker.SessionInfo{SessionID:"audit-session",UserID:"audit-user"}))
    require.Equal(t,2,m.refs)
    ctx,cancel := context.WithCancel(context.Background())
    cancel()
    _,err := w.ResetContext(ctx)
    require.ErrorIs(t,err,context.Canceled)
    require.NoError(t,w.Terminate(context.Background()))
    require.NoError(t,w.Terminate(context.Background()))
    require.Equal(t,1,m.refs,"failed reset leaked this Worker's acquired manager reference")
}

func TestAudit987OwnershipManagerCloseThenConnectionClose(t *testing.T) {
    t.Parallel()
    m := auditRPCManager()
    c := &appConn{recvCh:m.Subscribe("audit-thread","audit-session"),manager:m}
    m.Shutdown(context.Background())
    require.NotPanics(t,func(){require.NoError(t,c.Close())})
}

func TestAudit987OwnershipConnectionTrySendAfterClose(t *testing.T) {
    t.Parallel()
    c := &appConn{recvCh:make(chan *events.Envelope,1)}
    require.NoError(t,c.Close())
    require.NotPanics(t,func(){require.False(t,c.TrySend(&events.Envelope{}))})
}
